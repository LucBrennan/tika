        when(registry.getTypes()).thenReturn(new TreeSet<>());
        when(registry.getTypes()).thenReturn(new TreeSet<>(List.of(mimeType)));
        when(mimeType.getName()).thenReturn("example");
        assertEquals(mimeType, patterns.matches("example.txt"));
    }

    @Test

    public void testMatchesNullName() {
        assertThrows(IllegalArgumentException.class, () -> patterns.matches(null));
